    }

sealed class PixelShader : IDisposable {
    public static int Live;
    bool disposed;
    public PixelShader() { Live++; }
    public void Dispose() { if (disposed) throw new Exception("double shader disposal"); disposed = true; Live--; }
}

sealed class TextureManager {
    public readonly HashSet<IRtvTexture> Live = new();
    public int Calls, FailAt;
    public IRtvTexture CreateRtv(string name, int w, int h, Format format) {
        if (++Calls == FailAt) throw new InvalidOperationException("allocation fault");
        var texture = new IRtvTexture { Format = format }; Live.Add(texture); return texture;
    }
    public void DisposeTex(ref IRtvTexture texture) {
        if (texture == null) return;
        var retired = texture;
        if (AnomalyHook.Publications.Any(p => p.IsAvailable && ReferenceEquals(p.Srv, retired))) throw new Exception("disposed publication is still available");
        if (!Live.Remove(texture)) throw new Exception("double/unowned texture disposal");
        texture = null;
    }
}
sealed class BufferManager {
    public readonly HashSet<IConstantBuffer> Live = new();
    public IConstantBuffer CreateConstantBuffer(string name, int size, ResourceUsage usage, bool isGlobal) { var buffer = new IConstantBuffer(); Live.Add(buffer); return buffer; }
    public void Dispose(IConstantBuffer buffer) { if (!Live.Remove(buffer)) throw new Exception("double buffer disposal"); }
}
